def is_daily_quota_exhaustion(exc: Exception) -> bool:
    """Detect if 429 error is due to daily token/request exhaustion (TPD / tokens per day only)."""
    msg = str(exc).lower()
    return "tokens per day" in msg or "(tpd)" in msg or " tpd" in msg


def _extract_retry_after(exc: Exception) -> Optional[float]:
    """
    Extract Retry-After header or message in seconds from 429 response if present.
    Strictly anchored on 'try again in|retry after|wait' to avoid matching arbitrary numbers like 'Limit 200000'.
    Caps sleep at 20.0s.
    Returns None if daily quota is exhausted.
    """
    if is_daily_quota_exhaustion(exc):
        return None

    msg = str(exc)
    # Check for direct headers attribute if available on HTTP response
    response = getattr(exc, "response", None)
    if response is not None:
        headers = getattr(response, "headers", {})
        retry_after = headers.get("retry-after") or headers.get("Retry-After")
        if retry_after:
            try:
                val = float(retry_after)
                return min(val, 20.0)
            except (ValueError, TypeError):
                pass
                
    # Search message strictly anchored on 'try again in', 'retry after', or 'wait'
    match = re.search(r"(?:try again in|retry after|wait)\s+([\d\.]+)\s*(ms|s|seconds)?", msg, re.IGNORECASE)
    if match:
        try:
            val = float(match.group(1))
            unit = (match.group(2) or "s").lower()
            if unit == "ms":
                seconds = val / 1000.0
            else:
                seconds = val
            return min(seconds, 20.0)
        except (ValueError, TypeError):
            pass
    return None

import sys, os
sys.path.insert(0, os.path.abspath("."))

import pytest
from app.gateway.client import _extract_retry_after, is_retryable_exception, is_daily_quota_exhaustion


class MockResponse:
    def __init__(self, headers):
        self.headers = headers


class MockException(Exception):
    def __init__(self, message, headers=None, status_code=429):
        super().__init__(message)
        self.status_code = status_code
        if headers is not None:
            self.response = MockResponse(headers)


def test_extract_retry_after_ms():
    exc = Exception("Rate limit reached. Please try again in 797.14ms. Visit https://...")
    res = _extract_retry_after(exc)
    assert res is not None
    assert pytest.approx(res, rel=1e-3) == 0.79714


def test_extract_retry_after_seconds():
    exc = Exception("Rate limit exceeded. Please try again in 2.5s.")
    res = _extract_retry_after(exc)
    assert res == 2.5


def test_extract_retry_after_header_within_cap():
    exc = MockException("Rate limit", headers={"retry-after": "12.0"})
    res = _extract_retry_after(exc)
    assert res == 12.0


def test_extract_retry_after_header_capped_at_20s():
    exc = MockException("Rate limit", headers={"retry-after": "45.0"})
    res = _extract_retry_after(exc)
    assert res == 20.0


def test_extract_retry_after_limit_number_does_not_set_sleep():
    # A message containing 'Limit 200000' must not set the sleep to 200000; it must extract 1.2s
    exc = Exception("Rate limit reached. Limit 200000, Used 199000. Please try again in 1.2s.")
    res = _extract_retry_after(exc)
    assert res == 1.2


def test_extract_retry_after_daily_limit_tpd_fails_fast():
    # When tokens per day (TPD) is mentioned, fail fast and return None
    exc = Exception("Rate limit reached for model `qwen/qwen3.8-27b` on tokens per day (TPD): Limit 200000, Used 200050.")
    res = _extract_retry_after(exc)
    assert res is None


def test_daily_limit_fails_fast_not_retryable():
    exc = MockException(
        "Rate limit reached for model on tokens per day (TPD): Limit 200000, Used 200050. Please try again tomorrow.",
        status_code=429
    )
    assert is_daily_quota_exhaustion(exc) is True
    assert is_retryable_exception(exc) is False


def test_transient_rate_limit_is_retryable():
    exc = MockException(
        "Rate limit reached. Please try again in 500ms.",
        status_code=429
    )
    assert is_daily_quota_exhaustion(exc) is False
    assert is_retryable_exception(exc) is True
